DATABASE SCHEMA
═══════════════════════════════════════════════════════════════════════════════

conversations:
  id (TEXT PRIMARY KEY)
  title (TEXT)
  platforms (TEXT - JSON array)
  avatar (TEXT)
  unread_count (INTEGER)
  created_at (DATETIME)
  updated_at (DATETIME)
  INDEX: updated_at

messages:
  id (TEXT PRIMARY KEY)
  conversation_id (TEXT FK → conversations.id)
  platform (TEXT - 'whatsapp', 'telegram', 'ai')
  sender_id (TEXT)
  sender_name (TEXT)
  sender_avatar (TEXT)
  content (TEXT)
  timestamp (DATETIME)
  status (TEXT - 'sending', 'sent', 'delivered', 'read', 'error')
  metadata (TEXT - JSON)
  INDEX: conversation_id, timestamp

users:
  id (TEXT PRIMARY KEY)
  username (TEXT UNIQUE)
  avatar (TEXT)
  created_at (DATETIME)

contacts:
  id (TEXT PRIMARY KEY)
  user_id (TEXT FK → users.id)
  contact_id (TEXT)
  platform (TEXT)
  name (TEXT)
  avatar (TEXT)
  INDEX: user_id

settings:
  key (TEXT PRIMARY KEY)
  value (TEXT)
  updated_at (DATETIME)

DEVELOPMENT COMMANDS
═══════════════════════════════════════════════════════════════════════════════

npm install

# Building

# Code Quality
npm run lint                         # Run ESLint
npm run format                       # Format with Prettier
npm run type-check                   # TypeScript check

# Backend
cd src-tauri
cargo build --release               # Build Rust binary
cargo test                          # Run tests
cargo clippy                        # Lint Rust code

QUICK START GUIDE
═══════════════════════════════════════════════════════════════════════════════

1. Install Dependencies:
   npm install

2. Setup Rust Backend:
   cd src-tauri && cargo fetch && cd ..

3. Start Development:
   npm run tauri dev

4. The app will open at http://localhost:1420 (dev mode)

5. Make changes to:
   - React components in src/
   - Rust code in src-tauri/src/
   - Styles in src/**/*.css
